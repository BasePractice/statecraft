# Дополнительные примеры курса, не привязанные к одной лекции.
#
# Каталог разделён по составляющим:
#   loader/     — сам погрузчик: система управления на Takt, планировщик
#                 маршрута, конфигурация цеха, модель установки, тесты;
#   loader-gui/ — графическая составляющая на C++ (raylib), необязательная.
#
# Консольная часть собирается всегда. Графическая — только при
# -DSTATECRAFT_BUILD_GUI=ON и найденной raylib: курс обязан собираться в CI,
# где ни графических библиотек, ни дисплея нет.
project(21-additional)

option(STATECRAFT_BUILD_GUI "Собирать графические приложения (нужна raylib)" OFF)

add_subdirectory(loader)
add_subdirectory(loader-gui)
